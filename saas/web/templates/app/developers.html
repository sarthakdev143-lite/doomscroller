Settings → Connectors → Add custom connector → {{ mcp_url }}. If your build can’t send a header, use the mcp-remote bridge in claude_desktop_config.json (needs Node.js):

claude_desktop_config.json
{
  "mcpServers": {
    "{{ mcp_name }}": {
      "command": "npx",
      "args": ["-y", "mcp-remote", "{{ mcp_url }}",
               "--header", "Authorization:Bearer ${CLIPPER_API_KEY}"],
      "env": { "CLIPPER_API_KEY": "dsk_live_YOUR_KEY" }
    }
  }
}
~/.cursor/mcp.json
{
  "mcpServers": {
    "{{ mcp_name }}": {
      "url": "{{ mcp_url }}",
      "headers": { "Authorization": "Bearer dsk_live_YOUR_KEY" }
    }
  }
}
.vscode/mcp.json
{
  "inputs": [{ "type": "promptString", "id": "clipper-key", "description": "API key", "password": true }],
  "servers": {
    "{{ mcp_name }}": {
      "type": "http",
      "url": "{{ mcp_url }}",
      "headers": { "Authorization": "Bearer ${input:clipper-key}" }
    }
  }
}
